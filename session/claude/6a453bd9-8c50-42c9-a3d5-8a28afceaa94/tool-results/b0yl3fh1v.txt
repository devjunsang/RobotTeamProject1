     1	// ARIS3 Task Manager: Pick & Place 작업 상태 머신 (ROS 2 액션 서버).
     2	//
     3	//   /aris3/execute_task (ExecuteTask 액션)
     4	//        │ goal: target_color, target_shape, destination, dry_run
     5	//        ▼
     6	//   VALIDATE → OBSERVE → PERCEIVE → PLAN → PICK → PLACE → VERIFY → 완료
     7	//        │ 어느 단계든 실패·취소·시간 초과 → RECOVER (정지 / 위로 피하기 / 홈) → 결과
     8	//
     9	// 입력  /aris3/detected_objects  (aris3_interfaces/DetectedObjectArray)  ← 비전
    10	// 출력  /aris3/task_status       (aris3_interfaces/TaskStatus)           → GUI·모니터링
    11	// 사용  /aris3/gripper/command   (aris3_interfaces/srv/GripperCommand)   → 그리퍼 어댑터
    12	//       <robot_ns>/motion/*      (dsr_msgs2)                             → 두산 컨트롤러(에뮬레이터)
    13	//       TF world → base_link                                              → 좌표 변환
    14	//
    15	// 스레드 구조:
    16	//   - 액션 콜백(handleGoal/handleCancel/handleAccepted)과 구독 콜백은 executor 스레드에서 빨리 끝난다.
    17	//   - 실제 작업(execute)은 별도 std::thread 에서 돈다. 서비스 응답은 executor 의 다른 스레드가 받는다.

   226	TaskManagerNode::TaskManagerNode(const rclcpp::NodeOptions & options)
   227	: rclcpp::Node("aris3_task_manager", options)
   228	{
   229	  loadParameters();
   230	
   231	  // 콜백 그룹: 서비스 응답은 여러 개가 동시에 처리돼도 되므로 Reentrant.
   232	  // 카메라 구독은 한 번에 하나씩(MutuallyExclusive) 처리한다.
   233	  client_group_ = create_callback_group(rclcpp::CallbackGroupType::Reentrant);
   234	  input_group_ = create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive);
   235	
   236	  RobotClient::Options ro;
   237	  ro.ns = get_parameter("robot_ns").as_string();
   238	  ro.linear_vel_mm_s = get_parameter("linear_vel_mm_s").as_double();
   239	  ro.linear_acc_mm_s2 = 2.0 * ro.linear_vel_mm_s;
   240	  ro.joint_vel_deg_s = get_parameter("joint_vel_deg_s").as_double();
   241	  ro.joint_acc_deg_s2 = ro.joint_vel_deg_s;
   242	  ro.motion_timeout_s = get_parameter("motion_timeout_s").as_double();
   243	  ro.arrival_tolerance_mm = get_parameter("arrival_tolerance_mm").as_double();
   244	  robot_ = std::make_unique<RobotClient>(this, client_group_, ro);
   245	  gripper_ = std::make_unique<GripperClient>(this, client_group_,
   246	      get_parameter("gripper_service").as_string(), get_parameter("gripper_timeout_s").as_double());
   247	
   248	  // TF: 별도 스레드가 /tf, /tf_static 을 받아 buffer 에 쌓는다.
   249	  tf_buffer_ = std::make_shared<tf2_ros::Buffer>(get_clock());
   250	  tf_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf_buffer_);
   251	
   252	  rclcpp::SubscriptionOptions sub_opts;
   253	  sub_opts.callback_group = input_group_;
   254	  detections_sub_ = create_subscription<DetectedObjectArray>(
   255	    get_parameter("detections_topic").as_string(), 10,
   256	    std::bind(&TaskManagerNode::onDetections, this, std::placeholders::_1), sub_opts);
   257	
   258	  status_pub_ = create_publisher<TaskStatus>("/aris3/task_status", 10);
   259	  status_timer_ = create_wall_timer(1s, std::bind(&TaskManagerNode::onStatusTimer, this));
   260	
   261	  // 중복 실행 방지: 같은 이름의 액션 서버가 둘이면 목표가 양쪽에 다 전달되어 두 프로세스가 같은 로봇을
   262	  // 동시에 움직인다 (2026-11-07 시뮬레이션에서 실제로 겪음).
   263	  const std::string action_name = get_parameter("action_name").as_string();
   264	  if (actionServerAlive(action_name)) {
   265	    throw std::runtime_error("같은 이름의 액션 서버(" + action_name + ")가 이미 실행 중 - 두 번 띄우지 마세요");
   266	  }
   267	
   268	  action_server_ = rclcpp_action::create_server<ExecuteTask>(
   269	    this, action_name,
   270	    std::bind(&TaskManagerNode::handleGoal, this, std::placeholders::_1, std::placeholders::_2),
   271	    std::bind(&TaskManagerNode::handleCancel, this, std::placeholders::_1),
   272	    std::bind(&TaskManagerNode::handleAccepted, this, std::placeholders::_1));
   273	
   274	  publishStatus(TaskStatus::STATE_IDLE, "", "", "작업 대기");
   275	  RCLCPP_INFO(get_logger(), "준비됨: 액션 %s, 비전 %s, 그리퍼 %s, 로봇 %s",
   276	    get_parameter("action_name").as_string().c_str(),
   277	    get_parameter("detections_topic").as_string().c_str(), gripper_->serviceName().c_str(), ro.ns.c_str());
   278	}
   279	
   280	// 이미 살아 있는 액션 서버가 있는가?
   281	// ROS 그래프에 이름이 보이는지만 보면 안 된다: 강제 종료된 프로세스의 이름이 잠시(수십 초) 남아 있어서
   282	// 정상적인 재시작까지 막는다(실제로 겪음). 그래서 "존재하지 않는 목표의 결과"를 실제로 물어본다.
   283	// 살아 있는 서버는 STATUS_UNKNOWN 으로 바로 답하고, 진행 중인 작업에는 아무 영향이 없다.
   284	bool TaskManagerNode::actionServerAlive(const std::string & action_name)
   285	{
   286	  using GetResult = ExecuteTask::Impl::GetResultService;
   287	  auto probe = std::make_shared<rclcpp::Node>("aris3_task_manager_probe");
   288	  auto client = probe->create_client<GetResult>(action_name + "/_action/get_result");
   289	  if (!client->wait_for_service(3s)) {   // 새 프로세스가 다른 노드를 발견하는 데 1~2 초 걸릴 수 있다
   290	    RCLCPP_INFO(probe->get_logger(), "중복 확인: %s 없음", action_name.c_str());
   291	    return false;   // 이름조차 없음
   292	  }
   293	  auto req = std::make_shared<GetResult::Request>();
   294	  for (size_t i = 0; i < req->goal_id.uuid.size(); ++i) {req->goal_id.uuid[i] = static_cast<uint8_t>(0xA5 ^ i);}
   295	  auto fut = client->async_send_request(req);
   296	  const auto rc = rclcpp::spin_until_future_complete(probe, fut, 1500ms);
   297	  RCLCPP_INFO(probe->get_logger(), "중복 확인: %s 이름 있음, 응답 %s", action_name.c_str(),
   298	    rc == rclcpp::FutureReturnCode::SUCCESS ? "있음(살아 있는 서버)" : "없음(이전 프로세스의 흔적)");
   299	  return rc == rclcpp::FutureReturnCode::SUCCESS;
   300	}

   309	void TaskManagerNode::loadParameters()
   310	{
   311	  // declare_parameter: 이름, 기본값. 실제 값은 config/task_manager.yaml 이 덮어쓴다.
   312	  declare_parameter("action_name", "/aris3/execute_task");
   313	  declare_parameter("robot_ns", "/dsr01");
   314	  declare_parameter("detections_topic", "/aris3/detected_objects");
   315	  declare_parameter("gripper_service", "/aris3/gripper/command");
   316	  declare_parameter("gripper_timeout_s", 10.0);
   317	  declare_parameter("world_frame", "workcell");   // 기본값은 config/task_manager.yaml 과 같게 유지
   318	  declare_parameter("robot_base_frame", "base_link");
   319	  declare_parameter("home_joints_deg", std::vector<double>{0.0, 0.0, 90.0, 0.0, 90.0, 0.0});
   320	  declare_parameter("tool_orientation_zyz_deg", std::vector<double>{0.0, 180.0, 0.0});
   321	  declare_parameter("tcp_offset_m", 0.08);
   322	  declare_parameter("grasp_offset_m", 0.005);
   323	  declare_parameter("approach_height_m", 0.10);
   324	  declare_parameter("object_height_m", 0.04);
   325	  declare_parameter("place_clearance_m", 0.002);
   326	  declare_parameter("workspace_r_min_m", 0.20);
   327	  declare_parameter("workspace_r_max_m", 0.75);
   328	  declare_parameter("workspace_z_min_m", 0.005);
   329	  declare_parameter("workspace_z_max_m", 0.60);
   330	  declare_parameter("zone_names", std::vector<std::string>{"A", "B", "C"});
   331	  declare_parameter("zone_xy_world", std::vector<double>{0.30, 0.35, 0.45, 0.35, 0.60, 0.35});
   332	  declare_parameter("table_z_world", 0.75);
   333	  declare_parameter("zone_tolerance_m", 0.03);
   334	  declare_parameter("zone_clear_radius_m", 0.06);
   335	  declare_parameter("known_colors", std::vector<std::string>{"red", "green", "blue"});
   336	  declare_parameter("perception_frames", 5);
   337	  declare_parameter("perception_timeout_s", 5.0);
   338	  declare_parameter("max_spread_m", 0.005);
   339	  declare_parameter("min_confidence", 0.5);
   340	  declare_parameter("settle_time_s", 0.5);
   341	  declare_parameter("linear_vel_mm_s", 100.0);
   342	  declare_parameter("joint_vel_deg_s", 40.0);
   343	  declare_parameter("motion_timeout_s", 30.0);
   344	  declare_parameter("arrival_tolerance_mm", 3.0);
   345	  declare_parameter("reach_tolerance_mm", 1.0);
   346	  declare_parameter("task_timeout_s", 180.0);
   347	  declare_parameter("grip_force", 0.5);
   348	  declare_parameter("max_grasp_attempts", 2);
   349	  declare_parameter("return_home_on_failure", true);
   350	
   351	  world_frame_ = get_parameter("world_frame").as_string();
   352	  base_frame_ = get_parameter("robot_base_frame").as_string();
   353	  const auto home = get_parameter("home_joints_deg").as_double_array();
   354	  const auto tool = get_parameter("tool_orientation_zyz_deg").as_double_array();
   355	  if (home.size() != 6 || tool.size() != 3) {
   356	    throw std::runtime_error("home_joints_deg 는 6개, tool_orientation_zyz_deg 는 3개여야 함");
   357	  }
   358	  std::copy(home.begin(), home.end(), home_deg_.begin());
   359	  std::copy(tool.begin(), tool.end(), tool_zyz_deg_.begin());
   360	  geom_.tcp_offset_m = get_parameter("tcp_offset_m").as_double();
   361	  geom_.grasp_offset_m = get_parameter("grasp_offset_m").as_double();
   362	  geom_.approach_height_m = get_parameter("approach_height_m").as_double();
   363	  geom_.object_height_m = get_parameter("object_height_m").as_double();
   364	  geom_.place_clearance_m = get_parameter("place_clearance_m").as_double();
   365	  workspace_.r_min_m = get_parameter("workspace_r_min_m").as_double();
   366	  workspace_.r_max_m = get_parameter("workspace_r_max_m").as_double();
   367	  workspace_.z_min_m = get_parameter("workspace_z_min_m").as_double();
   368	  workspace_.z_max_m = get_parameter("workspace_z_max_m").as_double();
   369	
   370	  const auto names = get_parameter("zone_names").as_string_array();
   371	  const auto xy = get_parameter("zone_xy_world").as_double_array();
   372	  const double table_z = get_parameter("table_z_world").as_double();
   373	  if (xy.size() != 2 * names.size()) {
   374	    throw std::runtime_error("zone_xy_world 는 zone_names 개수의 2배 (x, y 쌍) 여야 함");
   375	  }
   376	  for (size_t i = 0; i < names.size(); ++i) {
   377	    zones_world_[upper(names[i])] = Vec3{xy[2 * i], xy[2 * i + 1], table_z};
   378	  }
   379	  // 주의: get_parameter(...) 가 돌려주는 임시 객체의 배열을 바로 for 로 돌면, 반복 중에 임시 객체가
   380	  // 이미 사라져 있다(dangling reference). 먼저 지역 변수로 복사한다.
   381	  const auto colors = get_parameter("known_colors").as_string_array();
   382	  for (const auto & c : colors) {
   383	    known_colors_.push_back(lower(c));
   384	  }
   385	  zone_tolerance_m_ = get_parameter("zone_tolerance_m").as_double();
   386	  zone_clear_radius_m_ = get_parameter("zone_clear_radius_m").as_double();
   387	  perception_frames_ = static_cast<int>(get_parameter("perception_frames").as_int());
   388	  perception_timeout_s_ = get_parameter("perception_timeout_s").as_double();
   389	  max_spread_m_ = get_parameter("max_spread_m").as_double();
   390	  min_confidence_ = get_parameter("min_confidence").as_double();
   391	  settle_time_s_ = get_parameter("settle_time_s").as_double();
   392	  task_timeout_s_ = get_parameter("task_timeout_s").as_double();
   393	  reach_tolerance_mm_ = get_parameter("reach_tolerance_mm").as_double();
   394	  grip_force_ = get_parameter("grip_force").as_double();
   395	  max_grasp_attempts_ = static_cast<int>(get_parameter("max_grasp_attempts").as_int());
   396	  return_home_on_failure_ = get_parameter("return_home_on_failure").as_bool();
   397	}

   406	  if (busy_.exchange(true)) {

   413	    goal->dry_run ? " (dry_run)" : "");
   414	  return rclcpp_action::GoalResponse::ACCEPT_AND_EXECUTE;

   424	{

   431	// =====================================================================================
   432	// 작업 스레드: 상태 머신 본체
   433	// =====================================================================================
   434	void TaskManagerNode::execute(std::shared_ptr<GoalHandle> goal_handle)
   435	{
   436	  TaskContext ctx;
   437	  ctx.goal_handle = goal_handle;
   438	  ctx.goal = *goal_handle->get_goal();
   439	  ctx.start = std::chrono::steady_clock::now();
   440	  ctx.task_id = lower(ctx.goal.target_color) + "->" + upper(ctx.goal.destination) + " #" +
   441	    std::to_string(++task_counter_);
   442	
   443	  ErrorCode code = ErrorCode::NONE;
   444	  std::string message;
   445	  try {
   446	    stageValidate(ctx);
   447	    stageObserve(ctx);
   448	    stagePerceive(ctx);
   449	    stagePlan(ctx);
   450	    if (ctx.goal.dry_run) {
   451	      message = "dry_run: 인식과 계획까지 성공 (로봇은 움직이지 않음). 물체 " + xyz(ctx.object_world);
   452	    } else {
   453	      // 집기는 실패하면 정해진 횟수만큼 "다시 보고, 다시 계획하고" 재시도한다.
   454	      for (ctx.grasp_attempt = 1;; ++ctx.grasp_attempt) {
   455	        try {
   456	          stagePick(ctx);
   457	          break;
   458	        } catch (const TaskError & e) {
   459	          if (e.code() != ErrorCode::GRASP_FAILED || ctx.grasp_attempt >= max_grasp_attempts_) {
   460	            throw;
   461	          }
   462	          RCLCPP_WARN(get_logger(), "집기 실패 (%d/%d): %s -> 다시 관찰", ctx.grasp_attempt,
   463	            max_grasp_attempts_, e.what());
   464	          stageObserve(ctx);
   465	          stagePerceive(ctx);
   466	          stagePlan(ctx);
   467	        }
   468	      }
   469	      stagePlace(ctx);
   470	      stageVerify(ctx);
   471	      message = fmt("완료: 구역 중심에서 %.1f mm (허용 %.0f mm)", ctx.result->place_error_mm,
   472	          zone_tolerance_m_ * 1000, 0);
   473	    }
   474	  } catch (const TaskError & e) {
   475	    code = e.code();
   476	    message = e.what();
   477	  } catch (const std::exception & e) {
   478	    code = ErrorCode::INTERNAL;
   479	    message = std::string("예상하지 못한 오류: ") + e.what();
   480	  }
   481	
   482	  auto & r = *ctx.result;
   483	  const Stage failed_stage = ctx.stage;
   484	  if (code != ErrorCode::NONE) {
   485	    RCLCPP_ERROR(get_logger(), "[%s] 실패 %s: %s", stageName(failed_stage), errorName(code), message.c_str());
   486	    recover(ctx, code);
   487	    if (ctx.holding) {
   488	      message += " (물체를 잡은 상태로 끝남)";
   489	    }
   490	  }
   491	  r.success = (code == ErrorCode::NONE);
   492	  r.error_code = static_cast<uint8_t>(code);
   493	  r.message = message;
   494	  r.failed_stage = r.success ? "" : stageName(failed_stage);
   495	  r.duration_s = std::chrono::duration<float>(std::chrono::steady_clock::now() - ctx.start).count();
   496	
   497	  try {
   498	    if (code == ErrorCode::CANCELED && goal_handle->is_canceling()) {
   499	      goal_handle->canceled(ctx.result);
   500	    } else if (r.success) {
   501	      goal_handle->succeed(ctx.result);
   502	    } else {
   503	      goal_handle->abort(ctx.result);
   504	    }
   505	  } catch (const std::exception & e) {
   506	    RCLCPP_ERROR(get_logger(), "결과 전송 실패: %s", e.what());
   507	  }
   508	  RCLCPP_INFO(get_logger(), "작업 %s 종료: %s (%.1f s) %s", ctx.task_id.c_str(),
   509	    r.success ? "성공" : errorName(code), r.duration_s, message.c_str());
   510	  publishStatus(r.success ? TaskStatus::STATE_IDLE : TaskStatus::STATE_ERROR, ctx.task_id,
   511	    r.success ? "DONE" : stageName(failed_stage), message);
   512	  busy_ = false;  // 맨 마지막에 풀어야 다음 요청이 이 작업과 겹치지 않는다
   513	}

   515	// ---- 1. VALIDATE: 요청 내용과 장비 준비 상태를 움직이기 전에 모두 확인한다.
   516	void TaskManagerNode::stageValidate(TaskContext & ctx)
   517	{
   518	  enterStage(ctx, Stage::VALIDATE, "요청과 장비 준비 상태 확인");
   519	  const auto & g = ctx.goal;
   520	  if (!g.task_type.empty() && lower(g.task_type) != "pick_place") {
   521	    throw TaskError(ErrorCode::INVALID_GOAL, "지원하지 않는 작업 종류: " + g.task_type);
   522	  }
   523	  const std::string color = lower(g.target_color);
   524	  if (std::find(known_colors_.begin(), known_colors_.end(), color) == known_colors_.end()) {
   525	    std::string list;
   526	    for (const auto & c : known_colors_) {list += c + " ";}
   527	    throw TaskError(ErrorCode::INVALID_GOAL, "모르는 색 '" + g.target_color + "' (가능: " + list + ")");
   528	  }
   529	  auto zone = zones_world_.find(upper(g.destination));
   530	  if (zone == zones_world_.end()) {
   531	    std::string list;
   532	    for (const auto & z : zones_world_) {list += z.first + " ";}
   533	    throw TaskError(ErrorCode::INVALID_GOAL, "모르는 구역 '" + g.destination + "' (가능: " + list + ")");
   534	  }
   535	  ctx.zone_world = zone->second;
   536	  ctx.query = TargetQuery{color, lower(g.target_shape), min_confidence_};
   537	
   538	  std::string missing;
   539	  if (!robot_->servicesReady(&missing)) {
   540	    throw TaskError(ErrorCode::NOT_READY, "로봇 서비스 없음: " + missing + " (시뮬레이션/로봇 접속 확인)");
   541	  }
   542	  const int state = robot_->robotState();
   543	  if (state != kStandby) {
   544	    throw TaskError(ErrorCode::NOT_READY, "로봇이 대기(STANDBY) 상태가 아님 (state=" + std::to_string(state) + ")");
   545	  }
   546	  if (!gripper_->ready()) {
   547	    throw TaskError(ErrorCode::NOT_READY, "그리퍼 서비스 없음: " + gripper_->serviceName());
   548	  }
   549	  // 이전 작업이 취소되어 물체를 잡은 채 끝났을 수 있다. 그 상태로 새 작업을 시작하면 안 된다.
   550	  const auto gs = gripper_->status();
   551	  if (!gs.ok) {
   552	    throw TaskError(ErrorCode::NOT_READY, "그리퍼 상태 확인 실패: " + gs.message);
   553	  }
   554	  if (gs.holding) {
   555	    throw TaskError(ErrorCode::NOT_READY, "그리퍼가 이미 " + (gs.object_id.empty() ? std::string("물체") : gs.object_id) +
   556	            " 를 잡고 있음 - 먼저 정리 (README의 '물체를 잡은 채 멈췄을 때')");
   557	  }
   558	  {
   559	    std::lock_guard<std::mutex> lk(frames_mutex_);
   560	    const double age = std::chrono::duration<double>(std::chrono::steady_clock::now() - last_frame_time_).count();
   561	    if (frame_seq_ == 0 || age > 2.0) {
   562	      throw TaskError(ErrorCode::NOT_READY, "카메라 인식 결과가 들어오지 않음 (" +
   563	              get_parameter("detections_topic").as_string() + ")");
   564	    }
   565	  }
   566	  std::string tf_error;
   567	  if (!refreshTransforms(&tf_error)) {
   568	    throw TaskError(ErrorCode::NOT_READY, "좌표 변환 없음: " + tf_error);
   569	  }
   570	  report(ctx, "확인 완료: 로봇 대기, 그리퍼·카메라·TF 정상. 구역 " + upper(g.destination) + " " +
   571	    xyz(ctx.zone_world), 1.0);
   572	}

   587	// ---- 3. PERCEIVE: 여러 장의 카메라 결과로 목표 물체 위치를 정한다.
   588	void TaskManagerNode::stagePerceive(TaskContext & ctx)
   589	{
   590	  enterStage(ctx, Stage::PERCEIVE, "카메라로 " + ctx.query.color + " 물체 찾기");
   591	  const Estimate e = perceive(ctx, "집기 전");
   592	  switch (e.status) {
   593	    case EstimateStatus::OK: break;
   594	    case EstimateStatus::NOT_FOUND: throw TaskError(ErrorCode::TARGET_NOT_FOUND, "목표를 못 찾음: " + e.reason);
   595	    case EstimateStatus::UNSTABLE: throw TaskError(ErrorCode::TARGET_UNSTABLE, "목표 위치 불안정: " + e.reason);
   596	    case EstimateStatus::AMBIGUOUS: throw TaskError(ErrorCode::TARGET_AMBIGUOUS, e.reason);
   597	  }
   598	  ctx.object_world = e.mean;
   599	  ctx.object_id = e.id;
   600	  ctx.result->picked_from = toPoint(e.mean);
   601	  // 놓을 구역이 비어 있는가? 다른 물체가 있으면 그 위에 쌓거나 부딪힌다.
   602	  const auto occupants = occupantsNear(ctx.last_objects, ctx.zone_world, zone_clear_radius_m_, e.mean);
   603	  if (!occupants.empty()) {
   604	    const auto & o = occupants.front();
   605	    throw TaskError(ErrorCode::DESTINATION_OCCUPIED, "구역 " + upper(ctx.goal.destination) + "에 이미 " +
   606	            (o.id.empty() ? o.color : o.id) + " 가 있음 " + xyz(o.position) + " - 움직이지 않음");
   607	  }
   608	  report(ctx, "찾음: " + (e.id.empty() ? ctx.query.color : e.id) + " " + xyz(e.mean) +
   609	    fmt(" (영상 %.0f장, 흔들림 %.1f mm)", e.frames_matched, e.spread_m * 1000, 0), 1.0);
   610	}
   611	
   612	Estimate TaskManagerNode::perceive(TaskContext & ctx, const std::string & purpose)
   613	{
   614	  TargetEstimator est(ctx.query);
   615	  uint64_t last_seq;
   616	  {
   617	    std::lock_guard<std::mutex> lk(frames_mutex_);
   618	    last_seq = frame_seq_;  // 지금 이후에 들어오는 영상만 쓴다 (이동 중에 찍힌 영상 제외)
   619	  }
   620	  const auto deadline = std::chrono::steady_clock::now() + std::chrono::duration<double>(perception_timeout_s_);
   621	  while (std::chrono::steady_clock::now() < deadline) {
   622	    const ErrorCode why = abortReason(ctx);
   623	    if (why != ErrorCode::NONE) {
   624	      throw TaskError(why, purpose + " 인식 중 중단");
   625	    }
   626	    {
   627	      std::lock_guard<std::mutex> lk(frames_mutex_);
   628	      for (const auto & f : frames_) {
   629	        if (f.seq > last_seq) {
   630	          est.addFrame(f.objects);
   631	          last_seq = f.seq;
   632	          ctx.last_objects = f.objects;
   633	        }
   634	      }
   635	    }
   636	    if (est.framesMatched() >= perception_frames_ || est.framesSeen() >= 3 * perception_frames_) {
   637	      break;
   638	    }
   639	    std::this_thread::sleep_for(50ms);
   640	  }
   641	  return est.estimate(perception_frames_, max_spread_m_);
   642	}

   644	// ---- 4. PLAN: 경로점을 계산하고, 움직이기 전에 모두 갈 수 있는지 확인한다.
   645	void TaskManagerNode::stagePlan(TaskContext & ctx)
   646	{
   647	  enterStage(ctx, Stage::PLAN, "경로점 계산과 도달 가능성 확인");
   648	  const Vec3 object_base = worldToBase(ctx.object_world);
   649	  const Vec3 zone_base = worldToBase(ctx.zone_world);
   650	  ctx.plan = makePickPlacePlan(object_base, zone_base, geom_);
   651	  const auto wps = ctx.plan.waypoints();
   652	  for (size_t i = 0; i < wps.size(); ++i) {
   653	    const auto & wp = wps[i];
   654	    const std::string why = checkWorkspace(wp.tcp, workspace_);
   655	    if (!why.empty()) {
   656	      throw TaskError(ErrorCode::UNREACHABLE, wp.name + " " + xyz(wp.tcp) + ": " + why + " - 움직이지 않음");
   657	    }
   658	    const double err = robot_->reachabilityErrorMm(flangePoseMmDeg(wp.tcp, geom_, tool_zyz_deg_));
   659	    if (err < 0.0) {
   660	      throw TaskError(ErrorCode::NOT_READY, "역기구학(ikin/fkin) 서비스 응답 없음");
   661	    }
   662	    if (err > reach_tolerance_mm_) {
   663	      throw TaskError(ErrorCode::UNREACHABLE, wp.name + " " + xyz(wp.tcp) +
   664	              fmt(": 로봇이 닿지 않음 (IK->FK 오차 %.1f mm) - 움직이지 않음", err, 0, 0));
   665	    }
   666	    report(ctx, wp.name + " " + xyz(wp.tcp) + " (base) 확인", (i + 1.0) / wps.size());
   667	  }
   668	}

   670	// ---- 5. PICK: 열기 → 물체 위 → 하강 → 닫기(잡았는지 확인) → 들어 올리기
   671	void TaskManagerNode::stagePick(TaskContext & ctx)
   672	{
   673	  enterStage(ctx, Stage::PICK, "집기 (시도 " + std::to_string(ctx.grasp_attempt) + "/" +
   674	    std::to_string(max_grasp_attempts_) + ")");
   675	  auto g = gripper_->open();
   676	  if (!g.ok) {
   677	    throw TaskError(ErrorCode::MOTION_FAILED, "그리퍼 열기 실패: " + g.message);
   678	  }
   679	  moveTcp(ctx, ctx.plan.above_object, "물체 위로");
   680	  moveTcp(ctx, ctx.plan.grasp, "집기 위치로 하강");
   681	  report(ctx, "그리퍼 닫기", 0.6);
   682	  g = gripper_->close(grip_force_);
   683	  const bool wrong_object = !ctx.object_id.empty() && !g.object_id.empty() && g.object_id != ctx.object_id;
   684	  if (!g.ok || !g.holding || wrong_object) {
   685	    gripper_->open();
   686	    moveTcp(ctx, ctx.plan.above_object, "빈손으로 위로");
   687	    throw TaskError(ErrorCode::GRASP_FAILED, "잡지 못함: " + g.message +
   688	            (wrong_object ? " (다른 물체 " + g.object_id + ")" : ""));
   689	  }
   690	  ctx.holding = true;
   691	  report(ctx, "잡음: " + (g.object_id.empty() ? ctx.query.color : g.object_id), 0.7);
   692	  moveTcp(ctx, ctx.plan.above_object, "들어 올리기");
   693	}

   695	// ---- 6. PLACE: 구역 위 → 하강 → 열기 → 위로 빠지기
   696	void TaskManagerNode::stagePlace(TaskContext & ctx)
   697	{
   698	  enterStage(ctx, Stage::PLACE, "구역 " + upper(ctx.goal.destination) + "에 놓기");
   699	  moveTcp(ctx, ctx.plan.above_zone, "구역 위로");
   700	  moveTcp(ctx, ctx.plan.place, "놓는 위치로 하강");
   701	  auto g = gripper_->open();
   702	  if (!g.ok) {
   703	    throw TaskError(ErrorCode::MOTION_FAILED, "그리퍼 열기 실패: " + g.message);
   704	  }
   705	  ctx.holding = false;
   706	  report(ctx, "놓음", 0.7);
   707	  moveTcp(ctx, ctx.plan.above_zone, "위로 빠지기");
   708	}

   710	// ---- 7. VERIFY: 홈으로 돌아가 카메라로 물체가 구역 안에 있는지 다시 확인 (닫힌 고리)
   711	void TaskManagerNode::stageVerify(TaskContext & ctx)
   712	{
   713	  enterStage(ctx, Stage::VERIFY, "관찰 자세로 돌아가 카메라로 결과 확인");
   714	  robot_->moveJoints(home_deg_, "관찰 자세(홈)", abortCheck(ctx));
   715	  sleepChecked(ctx, settle_time_s_);
   716	  const Estimate e = perceive(ctx, "놓은 뒤");
   717	  if (e.status != EstimateStatus::OK) {
   718	    throw TaskError(ErrorCode::PLACE_VERIFY_FAILED, "놓은 뒤 물체를 확인하지 못함: " + e.reason);
   719	  }
   720	  const double err = normXY(e.mean - ctx.zone_world);
   721	  ctx.result->placed_at = toPoint(e.mean);
   722	  ctx.result->place_error_mm = static_cast<float>(err * 1000.0);
   723	  if (err > zone_tolerance_m_) {
   724	    throw TaskError(ErrorCode::PLACE_VERIFY_FAILED,
   725	            fmt("구역 밖: 중심에서 %.1f mm (허용 %.0f mm)", err * 1000, zone_tolerance_m_ * 1000, 0));
   726	  }
   727	  report(ctx, "구역 안 확인: 물체 " + xyz(e.mean) + fmt(", 중심에서 %.1f mm", err * 1000, 0, 0), 1.0);
   728	}
   729	
   730	// ---- 실패·취소 후 처리
   731	// 취소: 정지만 한다. 사용자가 "멈춰" 라고 했는데 로봇이 더 움직이면 놀란다.
   732	// 실패: 정지 → (낮은 곳에 있으면) 위로 피하기 → 홈. 물체를 잡고 있으면 놓지 않는다(떨어뜨리지 않기).
   733	void TaskManagerNode::recover(TaskContext & ctx, ErrorCode cause)
   734	{
   735	  const Stage failed = ctx.stage;
   736	  ctx.stage = Stage::RECOVER;
   737	  report(ctx, std::string("정지 (원인 ") + errorName(cause) + ")");
   738	  robot_->stop();
   739	  if (cause == ErrorCode::CANCELED) {
   740	    report(ctx, ctx.holding ? "취소: 정지만 함 (물체는 잡은 채)" : "취소: 정지만 함");
   741	    return;
   742	  }
   743	  const bool arm_may_be_low = (failed == Stage::PICK || failed == Stage::PLACE);
   744	  if (!return_home_on_failure_ || !arm_may_be_low || ctx.goal.dry_run) {
   745	    return;
   746	  }
   747	  // 복구 이동에는 취소를 적용하지 않고 이동 시간 초과만 적용한다.
   748	  const RobotClient::AbortCheck only_timeout = []() {return ErrorCode::NONE;};
   749	  try {
   750	    if (auto pose = robot_->currentPose()) {
   751	      Vec3 tcp = tcpFromFlangeMm(*pose, geom_);
   752	      const double safe_z = std::min(tcp.z + geom_.approach_height_m, workspace_.z_max_m);
   753	      if (tcp.z < safe_z - 0.005) {
   754	        tcp.z = safe_z;
   755	        report(ctx, "복구: 위로 피하기");
   756	        robot_->moveLinear(flangePoseMmDeg(tcp, geom_, tool_zyz_deg_), "복구: 위로", only_timeout);
   757	      }
   758	    }
   759	    report(ctx, "복구: 홈으로");
   760	    robot_->moveJoints(home_deg_, "복구: 홈", only_timeout);
   761	    report(ctx, "복구 완료");
   762	  } catch (const TaskError & e) {
   763	    RCLCPP_ERROR(get_logger(), "복구 이동 실패: %s (사람이 확인해야 함)", e.what());
   764	  }
   765	}

   777	void TaskManagerNode::enterStage(TaskContext & ctx, Stage stage, const std::string & detail)
   778	{
   779	  ctx.stage = stage;
   780	  ctx.stage_fraction = 0.0;
   781	  publishStatus(TaskStatus::STATE_BUSY, ctx.task_id, stageName(stage), detail);
   782	  report(ctx, detail, 0.0);
   783	}

   785	void TaskManagerNode::report(TaskContext & ctx, const std::string & detail, double stage_fraction)
   786	{
   787	  auto fb = std::make_shared<ExecuteTask::Feedback>();
   788	  const int index = std::min(static_cast<int>(ctx.stage), kMainStageCount);
   789	  fb->stage = stageName(ctx.stage);
   790	  fb->stage_index = static_cast<uint8_t>(index);
   791	  fb->stage_count = kMainStageCount;
   792	  fb->detail = detail;
   793	  if (stage_fraction >= 0.0) {  // 음수 = 진행률은 그대로 두고 설명만 보냄
   794	    ctx.stage_fraction = std::max(ctx.stage_fraction, std::min(1.0, stage_fraction));
   795	  }
   796	  fb->progress = static_cast<float>(std::min(1.0, (index + ctx.stage_fraction) / kMainStageCount));
   797	  RCLCPP_INFO(get_logger(), "[%s] %s", fb->stage.c_str(), detail.c_str());
   798	  try {
   799	    if (ctx.goal_handle->is_active()) {
   800	      ctx.goal_handle->publish_feedback(fb);
   801	    }
   802	  } catch (const std::exception &) {
   803	    // 결과를 이미 보낸 뒤라면 feedback 은 보낼 수 없다 — 무시
   804	  }
   805	}

   807	ErrorCode TaskManagerNode::abortReason(const TaskContext & ctx) const
   808	{
   809	  if (!rclcpp::ok()) {return ErrorCode::CANCELED;}
   810	  if (ctx.goal_handle->is_canceling()) {return ErrorCode::CANCELED;}
   811	  const double elapsed = std::chrono::duration<double>(std::chrono::steady_clock::now() - ctx.start).count();
   812	  if (elapsed > task_timeout_s_) {return ErrorCode::TIMEOUT;}
   813	  return ErrorCode::NONE;
   814	}

   816	RobotClient::AbortCheck TaskManagerNode::abortCheck(const TaskContext & ctx) const
   817	{
   818	  return [this, &ctx]() {return abortReason(ctx);};
   819	}

   821	void TaskManagerNode::sleepChecked(TaskContext & ctx, double seconds)
   822	{
   823	  const auto end = std::chrono::steady_clock::now() + std::chrono::duration<double>(seconds);
   824	  while (std::chrono::steady_clock::now() < end) {
   825	    const ErrorCode why = abortReason(ctx);
   826	    if (why != ErrorCode::NONE) {
   827	      throw TaskError(why, "대기 중 중단");
   828	    }
   829	    std::this_thread::sleep_for(50ms);
   830	  }
   831	}

   834	bool TaskManagerNode::refreshTransforms(std::string * error)
   835	{
   836	  for (int i = 0; i < 20; ++i) {   // 최대 2 초 기다린다 (TF 는 별도 스레드가 채워 준다)
   837	    try {
   838	      T_base_world_ = tf_buffer_->lookupTransform(base_frame_, world_frame_, tf2::TimePointZero);
   839	      T_world_base_ = tf_buffer_->lookupTransform(world_frame_, base_frame_, tf2::TimePointZero);
   840	      return true;
   841	    } catch (const tf2::TransformException & e) {
   842	      if (error) {*error = e.what();}
   843	      std::this_thread::sleep_for(100ms);
   844	    }
   845	  }
   846	  return false;
   847	}

   865	// ---- 카메라 결과 받기 (구독 콜백: 짧게 처리하고 버퍼에 넣기만 한다)
   866	void TaskManagerNode::onDetections(const DetectedObjectArray::SharedPtr msg)
   867	{
   868	  std::vector<Observation> objs;
   869	  objs.reserve(msg->objects.size());
   870	  for (const auto & o : msg->objects) {
   871	    const std::string frame = o.pose.header.frame_id.empty() ? msg->header.frame_id : o.pose.header.frame_id;
   872	    Vec3 p{o.pose.pose.position.x, o.pose.pose.position.y, o.pose.pose.position.z};
   873	    if (frame != world_frame_) {
   874	      try {
   875	        geometry_msgs::msg::PointStamped in, out;
   876	        in.header.frame_id = frame;
   877	        in.point = o.pose.pose.position;
   878	        out = tf_buffer_->transform(in, world_frame_);
   879	        p = {out.point.x, out.point.y, out.point.z};
   880	      } catch (const tf2::TransformException & e) {
   881	        RCLCPP_WARN_THROTTLE(get_logger(), *get_clock(), 5000, "검출 좌표 변환 실패 (%s -> %s): %s",
   882	          frame.c_str(), world_frame_.c_str(), e.what());
   883	        continue;
   884	      }
   885	    }
   886	    objs.push_back(Observation{o.id, o.color, o.shape, p, o.confidence});
   887	  }
   888	  std::lock_guard<std::mutex> lk(frames_mutex_);
   889	  frames_.push_back(Frame{++frame_seq_, std::move(objs)});
   890	  while (frames_.size() > kMaxBufferedFrames) {frames_.pop_front();}
   891	  last_frame_time_ = std::chrono::steady_clock::now();
   892	}
